Aula 02 · Telemetria de agentes

Instrumentação, multi-agentes e privacidade

Instrumentar à mão ensinou o padrão; agora ele vem pronto. Você troca os spans manuais do LLM pela auto-instrumentação do OpenTelemetry, separa o atendimento em dois agentes e propaga o contexto entre eles num único trace, mascara dados pessoais no Collector e decide quais traces guardar com tail sampling.

      terminal

"""Agente de atendimento da Loja: spans do LLM pela auto-instrumentação, spans do agente e das ferramentas à mão."""
import json, logging, os
from fastapi import FastAPI
from pydantic import BaseModel
from openai import OpenAI
from opentelemetry import trace
from opentelemetry.trace import Status, StatusCode
from opentelemetry.instrumentation.openai_v2 import OpenAIInstrumentor
from telemetria import configurar
from ferramentas import FERRAMENTAS, executar

configurar("atendimento")
OpenAIInstrumentor().instrument()
tracer = trace.get_tracer("loja.agente")
log = logging.getLogger("agente")

MODELO = os.getenv("LLM_MODEL", "openai/gpt-4.1-mini")
llm = OpenAI(base_url=os.getenv("LLM_BASE_URL"), api_key=os.getenv("LLM_API_KEY"))
SISTEMA = ("Você é o agente de atendimento da Loja. Responda em português, em no máximo 3 frases. "
           "Use consultar_pedido para pedidos e buscar_politica para trocas, devoluções, frete e garantia. "
           "Nunca invente prazos ou valores: use só o que as ferramentas retornarem. "
           "Se não conseguir resolver, use abrir_chamado.")

app = FastAPI()


class Pergunta(BaseModel):
    texto: str


def chamar_ferramenta(chamada):
    nome = chamada.function.name
    with tracer.start_as_current_span(f"execute_tool {nome}") as s:
        s.set_attribute("gen_ai.operation.name", "execute_tool")
        s.set_attribute("gen_ai.tool.name", nome)
        s.set_attribute("gen_ai.tool.call.id", chamada.id)
        try:
            return executar(nome, json.loads(chamada.function.arguments or "{}"))
        except Exception as e:
            s.record_exception(e)
            s.set_status(Status(StatusCode.ERROR, str(e)))
            return f"erro: {e}"


@app.post("/chat")
def chat(p: Pergunta):
    with tracer.start_as_current_span("invoke_agent atendimento") as span:
        span.set_attribute("gen_ai.operation.name", "invoke_agent")
        span.set_attribute("gen_ai.agent.name", "atendimento")
        log.info("pergunta recebida: %s", p.texto)
        msgs = [{"role": "system", "content": SISTEMA}, {"role": "user", "content": p.texto}]
        usadas = []
        for passo in range(1, 6):
            # o span "chat <modelo>" agora vem da auto-instrumentação
            m = llm.chat.completions.create(model=MODELO, messages=msgs, tools=FERRAMENTAS,
                                            temperature=0).choices[0].message
            if not m.tool_calls:
                span.set_attribute("agente.passos", passo)
                log.info("resposta em %d passos, ferramentas=%s", passo, usadas)
                return {"resposta": m.content, "ferramentas": usadas,
                        "trace_id": format(span.get_span_context().trace_id, "032x")}
            msgs.append(m.model_dump(exclude_none=True))
            for chamada in m.tool_calls:
                usadas.append(chamada.function.name)
                msgs.append({"role": "tool", "tool_call_id": chamada.id, "content": chamar_ferramenta(chamada)})
        span.set_status(Status(StatusCode.ERROR, "limite de passos"))
        log.warning("limite de passos atingido, ferramentas=%s", usadas)
        return {"resposta": "Não consegui concluir. Vou abrir um chamado.", "ferramentas": usadas}
EOF

    2.2

    Multi-agentes e propagação de contexto

    Conceito
    Capítulo 2 · Um agente chama outro

    Dois serviços, um único trace

    A Loja coloca um agente de triagem na frente: ele classifica a intenção e delega ao agente de atendimento. Sem propagação de contexto, seriam dois traces sem relação.

    W3C Trace Context

    http
    POST /chat
traceparent: 00-4bf92f3577b34da6a3ce929d0e0e4736-00f067aa0ba902b7-01
               │  └ trace_id                         └ span pai       └ amostrado
    • inject() escreve o cabeçalho na saída; o servidor faz extract() na entrada (o FastAPI já faz).
    • Vale para HTTP, filas, A2A e MCP: o contexto viaja junto da mensagem.
    • Resultado: o mapa de serviços do Tempo mostra triagem → atendimento.

    Referências: W3C Trace Context.

    1. Instale o cliente HTTP e crie o agente de triagem:

"""Agente de triagem: classifica a intenção com o LLM e delega ao agente de atendimento por HTTP."""
import logging, os
import httpx
from fastapi import FastAPI
from pydantic import BaseModel
from openai import OpenAI
from opentelemetry import trace
from opentelemetry.propagate import inject
from opentelemetry.instrumentation.openai_v2 import OpenAIInstrumentor
from telemetria import configurar

configurar("triagem")
OpenAIInstrumentor().instrument()
tracer = trace.get_tracer("loja.triagem")
log = logging.getLogger("triagem")
MODELO = os.getenv("LLM_MODEL", "openai/gpt-4.1-mini")
llm = OpenAI(base_url=os.getenv("LLM_BASE_URL"), api_key=os.getenv("LLM_API_KEY"))
app = FastAPI()


class Pergunta(BaseModel):
    texto: str


@app.post("/chat")
def chat(p: Pergunta):
    with tracer.start_as_current_span("invoke_agent triagem") as span:
        span.set_attribute("gen_ai.operation.name", "invoke_agent")
        span.set_attribute("gen_ai.agent.name", "triagem")
        intencao = llm.chat.completions.create(model=MODELO, temperature=0, messages=[
            {"role": "system", "content": "Classifique a mensagem em uma palavra: atendimento (pedidos, trocas, "
                                          "frete, garantia) ou outro. Responda só a palavra."},
            {"role": "user", "content": p.texto}]).choices[0].message.content.strip().lower()
        span.set_attribute("triagem.intencao", intencao)
        if "atendimento" not in intencao:
            log.info("fora do escopo: %s", intencao)
            return {"resposta": "Posso ajudar com pedidos, trocas, frete e garantia.", "agente": "triagem"}
        cabecalhos = {}
        inject(cabecalhos)  # escreve o traceparent: o próximo serviço continua o mesmo trace
        log.info("delegando ao atendimento com traceparent=%s", cabecalhos.get("traceparent"))
        r = httpx.post("http://localhost:8000/chat", json={"texto": p.texto}, headers=cabecalhos, timeout=120)
        return {**r.json(), "agente": "triagem → atendimento"}
EOF

    2.3

    Privacidade: dados pessoais na telemetria

    Conceito
    Capítulo 3 · O cliente digitou o CPF

    Telemetria também é dado pessoal

    Clientes escrevem CPF, e-mail e endereço na conversa. Se o prompt vai para logs e spans, a telemetria passa a ser dado pessoal sob a LGPD.

    Onde proteger

    Na aplicação

    • Não capturar conteúdo por padrão
    • Logar IDs, não textos

    No Collector

    • Processador transform / redaction
    • Uma regra vale para todos os serviços

    No backend

    • Retenção curta
    • Controle de acesso por time

    Referências: transform processor.

    Prática
    Exercício 2.3

    Mascarando CPF e e-mail no Collector

    ~20 min

    Adicionar regras de mascaramento no Collector e provar que o dado pessoal não chega ao Loki.

    1. Mande uma pergunta com dados pessoais e encontre-a no Loki ({service_name="atendimento"} |= "pergunta"):

      terminal
      pergunta "Meu CPF é 123.456.789-09 e meu e-mail ana@exemplo.com, qual o status do pedido 1002?"

    2. Reescreva o otelcol.yaml com o mascaramento e o tail sampling (usado no próximo exercício):

      terminal
      cat > ~/obs/otelcol.yaml <<'EOF'
receivers:
  otlp:
    protocols:
      grpc: { endpoint: 0.0.0.0:4317 }
      http: { endpoint: 0.0.0.0:4318 }
processors:
  batch: {}
  # Mascara CPF e e-mail antes de qualquer dado sair do Collector
  transform/pii:
    log_statements:
      - replace_pattern(log.body, "\\d{3}\\.\\d{3}\\.\\d{3}-\\d{2}", "***.***.***-**")
      - replace_pattern(log.body, "[\\w.+-]+@[\\w-]+\\.[\\w.]+", "<email>")
    trace_statements:
      - replace_all_patterns(span.attributes, "value", "\\d{3}\\.\\d{3}\\.\\d{3}-\\d{2}", "***.***.***-**")
  # Guarda 100% dos traces com erro ou lentos e só 25% do resto
  tail_sampling:
    decision_wait: 10s
    policies:
      - { name: erros, type: status_code, status_code: { status_codes: [ERROR] } }
      - { name: lentos, type: latency, latency: { threshold_ms: 8000 } }
      - { name: amostra, type: probabilistic, probabilistic: { sampling_percentage: 25 } }
exporters:
  otlphttp/lgtm:
    endpoint: http://lgtm:4318
  debug:
    verbosity: basic
service:
  pipelines:
    traces:  { receivers: [otlp], processors: [transform/pii, tail_sampling, batch], exporters: [otlphttp/lgtm, debug] }
    metrics: { receivers: [otlp], processors: [batch], exporters: [otlphttp/lgtm] }
    logs:    { receivers: [otlp], processors: [transform/pii, batch], exporters: [otlphttp/lgtm] }
EOF

    3. Reinicie o Collector e repita a pergunta:

      terminal
      cd ~/obs && docker compose restart otelcol && docker compose logs otelcol --tail 5
    Resultado esperado

    • Depois: o log mostra ***.***.***-** e <email>.
    O que aprendemos com este exercício?
    • Mascarar no Collector protege todos os serviços com uma regra só.
    • O ideal é não coletar; mascarar é a segunda linha de defesa.

    Exercício 2.4

    Tail sampling: erros e lentos sempre, o resto por amostra

    ~20 min

    Ver a política de tail sampling configurada no exercício anterior funcionando.

    1. Gere várias conversas (o script respeita o limite do GitHub Models):

      terminal
      cat > carga.py <<'EOF'
"""Gera conversas no agente em ritmo controlado (respeita o limite do GitHub Models)."""
import random, sys, time
import httpx

PERGUNTAS = [
    "Qual o status do pedido 1002?", "Meu pedido 1001 já foi entregue?", "Por que o pedido 1003 foi cancelado?",
    "Posso trocar um produto depois de 20 dias?", "Qual o prazo para devolução?", "O frete é grátis?",
    "Quanto tempo de garantia tem?", "Cadê meu pedido 9999?", "Quero falar com um humano sobre cobrança duplicada",
    "Paguei o pedido 1004 e ainda aparece aguardando pagamento",
]
total = int(sys.argv[1]) if len(sys.argv) > 1 else 20
intervalo = float(sys.argv[2]) if len(sys.argv) > 2 else 12
for i in range(1, total + 1):
    pergunta = random.choice(PERGUNTAS)
    inicio = time.time()
    try:
        r = httpx.post("http://localhost:8000/chat", json={"texto": pergunta}, timeout=120).json()
        print(f"{i:>3} {time.time() - inicio:5.1f}s {r.get('ferramentas')} {pergunta}")
        if random.random() < 0.5 and r.get("trace_id"):
            httpx.post("http://localhost:8000/feedback", json={"trace_id": r["trace_id"],
                                                               "util": random.random() < 0.8})
    except Exception as e:
        print(f"{i:>3} erro: {e}")
    time.sleep(intervalo)
EOF
      terminal
      python carga.py 10 6
